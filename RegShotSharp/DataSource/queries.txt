"CREATE TABLE \"SUCCESSES\" (\"ID\"	INTEGER NOT NULL UNIQUE, \"KEY\"	TEXT NOT NULL, \"PATH\" TEXT NOT NULL, \"VALUE_TYPE\"	TEXT NOT NULL, \"VALUE\"	TEXT, PRIMARY KEY(\"ID\" AUTOINCREMENT));",
"CREATE TABLE \"ERRORS\" (\"ID\"	INTEGER NOT NULL UNIQUE, \"KEY\"	TEXT NOT NULL, \"PATH\"	TEXT NOT NULL, PRIMARY KEY(\"ID\" AUTOINCREMENT));",
"CREATE TABLE \"KEYS\" (\"KEY\"	TEXT NOT NULL UNIQUE, PRIMARY KEY(\"KEY\"));"

"CREATE INDEX \"SUCCESSES_KEY_PATH\" ON \"SUCCESSES\" (\"KEY\", \"PATH\");",
"CREATE INDEX \"ERRORS_KEY_PATH\" ON \"ERRORS\" (\"KEY\", \"PATH\");"

SELECT 'REMOVED' AS EVENT, * FROM (
	SELECT KEY, PATH FROM SUCCESSES EXCEPT SELECT KEY, PATH FROM SUCCESSES2
)
UNION ALL
SELECT 'ADDED' AS EVENT, * FROM (
	SELECT KEY, PATH FROM SUCCESSES2 EXCEPT SELECT KEY, PATH FROM SUCCESSES
)

SELECT
	'EDITED' AS EVENT,
	s1.VALUE as old_value, s2.VALUE as new_value,
	s1.VALUE_TYPE as old_value_type, s2.VALUE_TYPE as new_value_type
FROM SUCCESSES2 s2
INNER JOIN SUCCESSES s1 ON s1.KEY = s2.KEY AND s1.PATH = s2.PATH
WHERE s1.VALUE IS NOT s2.VALUE OR s1.VALUE_TYPE IS NOT s2.VALUE_TYPE
